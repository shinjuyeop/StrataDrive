# Explicit, optional dependency experiment. No network fetch or default build dependency.
find_package(CycloneDDS 11.0.1 EXACT REQUIRED)
find_path(STRATADRIVE_ZENOH_INCLUDE zenoh.h REQUIRED)
find_library(STRATADRIVE_ZENOH_LIBRARY zenohc REQUIRED)
idlc_generate(TARGET probe_packet FILES packet.idl)
add_executable(dds_candidate_probe probe.cpp dds_channel.c)
target_include_directories(dds_candidate_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dds_candidate_probe PRIVATE owner_wire CycloneDDS::ddsc probe_packet)
add_executable(zenoh_candidate_probe probe.cpp zenoh_channel.c)
target_include_directories(zenoh_candidate_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${STRATADRIVE_ZENOH_INCLUDE})
target_link_libraries(zenoh_candidate_probe PRIVATE owner_wire ${STRATADRIVE_ZENOH_LIBRARY} Threads::Threads)
foreach(target dds_candidate_probe zenoh_candidate_probe)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
