cmake_minimum_required(VERSION 3.20)

project(StrataDrive LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(STRATADRIVE_SANITIZERS "Host 개발 검사에 ASan/UBSan 사용" OFF)
option(STRATADRIVE_TSAN "Host concurrency 검사에 ThreadSanitizer 사용" OFF)
if(STRATADRIVE_SANITIZERS AND STRATADRIVE_TSAN)
    message(FATAL_ERROR "ASan/UBSan and TSan require separate builds")
endif()
option(STRATADRIVE_TRAMPOLINE_PILOT "Build the opt-in Trampoline POSIX Steering bridge" OFF)
option(STRATADRIVE_SOMEIP_PILOT "Build the isolated Central/Front vSomeIP pilot" OFF)

function(stratadrive_quality target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(STRATADRIVE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
        if(STRATADRIVE_TSAN)
            target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=thread)
        endif()
    elseif(STRATADRIVE_SANITIZERS)
        message(FATAL_ERROR "STRATADRIVE_SANITIZERS requires GCC or Clang")
    endif()
endfunction()
