cmake_minimum_required(VERSION 3.20)

project(StrataDrive LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(STRATADRIVE_SANITIZERS "Host 개발 검사에 ASan/UBSan 사용" OFF)

function(stratadrive_quality target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(STRATADRIVE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    elseif(STRATADRIVE_SANITIZERS)
        message(FATAL_ERROR "STRATADRIVE_SANITIZERS requires GCC or Clang")
    endif()
endfunction()

add_library(steering_angle_codec STATIC interfaces/steering/steering_angle_codec.cpp)
target_include_directories(steering_angle_codec PUBLIC interfaces/steering)
stratadrive_quality(steering_angle_codec)

add_library(e2e_profile5 STATIC interfaces/common/e2e_profile5.cpp)
target_include_directories(e2e_profile5 PUBLIC interfaces/common)
stratadrive_quality(e2e_profile5)

add_library(steering_frame_codec STATIC interfaces/steering/steering_frame_codec.cpp)
target_link_libraries(steering_frame_codec PUBLIC steering_angle_codec e2e_profile5)
stratadrive_quality(steering_frame_codec)

add_library(steering_receiver STATIC interfaces/steering/steering_receiver.cpp)
target_link_libraries(steering_receiver PUBLIC steering_frame_codec)
stratadrive_quality(steering_receiver)

add_executable(steering_demo tools/steering_demo.cpp)
target_link_libraries(steering_demo PRIVATE steering_receiver)
stratadrive_quality(steering_demo)

include(CTest)
